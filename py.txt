# PYTHON COMMANDS

## Import Libraries

```python
import pandas as pd
import matplotlib.pyplot as plt
import seaborn as sns
import plotly.express as px
```

---

# 1. Scatter Plot

## Profit vs Sales

```python
plt.scatter(dataset['Profit'], dataset['Sales'])

plt.xlabel("Profit")
plt.ylabel("Sales")
plt.title("Profit vs Sales")

plt.show()
```

---

# 2. Line Plot

## Sales Over Time

```python
dataset['Order Date'] = pd.to_datetime(dataset['Order Date'])

dataset = dataset.sort_values('Order Date')

plt.plot(dataset['Order Date'], dataset['Sales'])

plt.xlabel("Order Date")
plt.ylabel("Sales")
plt.title("Sales Over Time")

plt.show()
```

---

# 3. Bar Plot

## Sales by Category

```python
sales_by_category = dataset.groupby('Category')['Sales'].sum()

plt.bar(sales_by_category.index, sales_by_category.values)

plt.xlabel("Category")
plt.ylabel("Sales")
plt.title("Sales by Category")

plt.show()
```

---

# 4. Histogram

# 6. Pie Chart

## Sales by Region

```python
region_sales = dataset.groupby('Region')['Sales'].sum()

plt.pie(
    region_sales.values,
    labels=region_sales.index,
    autopct='%1.1f%%'
)

plt.title("Sales by Region")

plt.show()
```

# 7. Seaborn Scatter Plot

```python

```

# 8. Linear Regression

```python
from sklearn.linear_model import LinearRegression

X = dataset[['Profit']]
y = dataset['Sales']

model = LinearRegression()

model.fit(X, y)

predictions = model.predict(X)

plt.scatter(X, y)
plt.plot(X, predictions)

plt.xlabel("Profit")
plt.ylabel("Sales")
plt.title("Linear Regression")

plt.show()
```

---

# 9. KMeans Clustering

```python
from sklearn.cluster import KMeans

kmeans = KMeans(n_clusters=3)

dataset['Cluster'] = kmeans.fit_predict(
    dataset[['Sales', 'Profit']]
)

print(dataset.head())
```

# 10. 3D Plot

```python
from mpl_toolkits.mplot3d import Axes3D

fig = plt.figure()

    dataset['Sales'],
```

# 12. Revenue by State

```python
state_sales = (
    dataset.groupby('State/Province')['Sales']
    .sum()
    .sort_values(ascending=False)
)

plt.show()
```

# 13. Top 5 Cities by Sales
